cmake_minimum_required(VERSION 3.19)
project(ricestep_pose_engine_tests LANGUAGES C CXX)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(RICESTEP_COMPONENTS_ROOT "${repo}/firmware/components")
add_subdirectory("${repo}/firmware/pose_runtime" pose_runtime)
# Instrumentation stays private to this host-only test build. The production
# CMake never defines it, so native firmware carries neither counters nor API.
target_compile_definitions(ricestep_pose PRIVATE RICESTEP_POINTWISE_TEST_STATS)
add_executable(pose_engine_test "${CMAKE_CURRENT_SOURCE_DIR}/../pose_engine_test.cc")
add_executable(pose_pointwise_test "${CMAKE_CURRENT_SOURCE_DIR}/../pose_pointwise_test.cc")
add_executable(pose_pointwise_engine_test "${CMAKE_CURRENT_SOURCE_DIR}/../pose_pointwise_engine_test.cc")
foreach(target pose_engine_test pose_pointwise_test pose_pointwise_engine_test)
    target_link_libraries(${target} PRIVATE ricestep_pose)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE RICESTEP_POINTWISE_TEST_STATS)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
endforeach()
